  <script>
    const canvas = document.getElementById("gameCanvas");
    const ctx = canvas.getContext("2d");

    const scoreEl = document.getElementById("scoreEl");
    const highScoreEl = document.getElementById("highScoreEl");
    const speedEl = document.getElementById("speedEl");
    const nitroBar = document.getElementById("nitroBar");
    const overlay = document.getElementById("overlay");
    const overlayTitle = document.getElementById("overlayTitle");
    const overlayDesc = document.getElementById("overlayDesc");
    const startBtn = document.getElementById("startBtn");

    const W = canvas.width;
    const H = canvas.height;
    const ROAD_LEFT = 46;
    const ROAD_RIGHT = W - 46;
    const ROAD_WIDTH = ROAD_RIGHT - ROAD_LEFT;
    const LANES = 4;
    const LANE_WIDTH = ROAD_WIDTH / LANES;

    // Âm thanh tổng hợp đơn giản (không cần file ngoài)
    let audioCtx = null;
    function playTone(freq, type, duration, vol = 0.08) {
      try {
        if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        if (audioCtx.state === "suspended") audioCtx.resume();
        const osc = audioCtx.createOscillator();
        const gain = audioCtx.createGain();
        osc.type = type;
        osc.frequency.setValueAtTime(freq, audioCtx.currentTime);
        gain.gain.setValueAtTime(vol, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
        osc.connect(gain);
        gain.connect(audioCtx.destination);
        osc.start();
        osc.stop(audioCtx.currentTime + duration);
      } catch (e) {}
    }

    let highScore = Number(localStorage.getItem("turbo_neon_highscore") || 0);
    highScoreEl.textContent = highScore;

    const keys = { left: false, right: false, nitro: false };

    window.addEventListener("keydown", (e) => {
      const k = e.key.toLowerCase();
      if (k === "arrowleft" || k === "a") keys.left = true;
      if (k === "arrowright" || k === "d") keys.right = true;
      if (k === "arrowup" || k === "w" || k === " ") keys.nitro = true;
    });

    window.addEventListener("keyup", (e) => {
      const k = e.key.toLowerCase();
      if (k === "arrowleft" || k === "a") keys.left = false;
      if (k === "arrowright" || k === "d") keys.right = false;
      if (k === "arrowup" || k === "w" || k === " ") keys.nitro = false;
    });

    function bindTouchButton(id, prop) {
      const el = document.getElementById(id);
      const onStart = (e) => { e.preventDefault(); keys[prop] = true; };
      const onEnd = (e) => { e.preventDefault(); keys[prop] = false; };
      el.addEventListener("touchstart", onStart, { passive: false });
      el.addEventListener("touchend", onEnd, { passive: false });
      el.addEventListener("mousedown", onStart);
      el.addEventListener("mouseup", onEnd);
      el.addEventListener("mouseleave", onEnd);
    }
    bindTouchButton("btnLeft", "left");
    bindTouchButton("btnRight", "right");
    bindTouchButton("btnNitro", "nitro");

    let gameState = "MENU"; // MENU | PLAYING | GAMEOVER
    let score = 0;
    let baseSpeed = 340;
    let roadOffset = 0;
    let spawnEnemyTimer = 0;
    let spawnPickupTimer = 0;
    let shakeTime = 0;

    const player = {
      x: W / 2 - 22,
      y: H - 150,
      w: 44,
      h: 82,
      vx: 0,
      maxSteer: 340,
      nitro: 100,
      tilt: 0
    };

    let enemies = [];
    let pickups = [];
    let particles = [];

    const ENEMY_COLORS = [
      { body: "#ef4444", roof: "#991b1b" },
      { body: "#f97316", roof: "#9a3412" },
      { body: "#10b981", roof: "#065f46" },
      { body: "#a855f7", roof: "#581c87" },
      { body: "#eab308", roof: "#854d0e" }
    ];

    function resetGame() {
      score = 0;
      baseSpeed = 340;
      roadOffset = 0;
      spawnEnemyTimer = 0;
      spawnPickupTimer = 0;
      shakeTime = 0;
      enemies = [];
      pickups = [];
      particles = [];

      player.x = W / 2 - player.w / 2;
      player.y = H - 150;
      player.vx = 0;
      player.nitro = 100;
      player.tilt = 0;

      scoreEl.textContent = "0";
      nitroBar.style.width = "100%";
      gameState = "PLAYING";
      overlay.classList.add("hidden");
      playTone(440, "triangle", 0.15, 0.1);
    }

    startBtn.addEventListener("click", resetGame);

    function getLaneCenterX(laneIndex, objWidth) {
      return ROAD_LEFT + laneIndex * LANE_WIDTH + (LANE_WIDTH - objWidth) / 2;
    }

    function spawnEnemy() {
      const lane = Math.floor(Math.random() * LANES);
      const w = 44;
      const h = 80;
      const x = getLaneCenterX(lane, w);

      // Tránh sinh xe đè sát xe vừa xuất hiện
      for (const e of enemies) {
        if (Math.abs(e.x - x) < 35 && e.y < 140) return;
      }

      const palette = ENEMY_COLORS[Math.floor(Math.random() * ENEMY_COLORS.length)];
      enemies.push({
        x,
        y: -h - 20,
        w,
        h,
        speedOffset: 90 + Math.random() * 110,
        color: palette
      });
    }

    function spawnPickup() {
      const lane = Math.floor(Math.random() * LANES);
      const isNitro = Math.random() < 0.28;
      const size = isNitro ? 28 : 24;
      const x = getLaneCenterX(lane, size);

      for (const e of enemies) {
        if (Math.abs(e.x - x) < 40 && e.y < 100) return;
      }

      pickups.push({
        x,
        y: -size - 10,
        w: size,
        h: size,
        type: isNitro ? "NITRO" : "COIN",
        pulse: 0
      });
    }

    function addExhaustParticle(x, y, isNitro) {
      particles.push({
        x: x + (Math.random() - 0.5) * 10,
        y,
        vx: (Math.random() - 0.5) * 30,
        vy: 120 + Math.random() * 90,
        life: 0.28,
        maxLife: 0.28,
        size: isNitro ? 7 : 4,
        color: isNitro ? "#38bdf8" : "#94a3b8"
      });
    }

    function addExplosion(x, y) {
      for (let i = 0; i < 36; i++) {
        const angle = Math.random() * Math.PI * 2;
        const spd = 60 + Math.random() * 260;
        particles.push({
          x,
          y,
          vx: Math.cos(angle) * spd,
          vy: Math.sin(angle) * spd,
          life: 0.65,
          maxLife: 0.65,
          size: 4 + Math.random() * 6,
          color: Math.random() < 0.5 ? "#f43f5e" : "#facc15"
        });
      }
    }

    function checkCollision(a, b, pad = 6) {
      return (
        a.x + pad < b.x + b.w - pad &&
        a.x + a.w - pad > b.x + pad &&
        a.y + pad < b.y + b.h - pad &&
        a.y + a.h - pad > b.y + pad
      );
    }

    function update(dt) {
      if (shakeTime > 0) shakeTime = Math.max(0, shakeTime - dt);

      // Cập nhật hạt hiệu ứng
      for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];
        p.x += p.vx * dt;
        p.y += p.vy * dt;
        p.life -= dt;
        if (p.life <= 0) particles.splice(i, 1);
      }

      if (gameState !== "PLAYING") return;

      // Tăng độ khó dần theo điểm
      baseSpeed = Math.min(760, 340 + score * 0.16);
      const isBoosting = keys.nitro && player.nitro > 2;
      const currentSpeed = isBoosting ? baseSpeed * 1.48 : baseSpeed;

      if (isBoosting) {
        player.nitro = Math.max(0, player.nitro - 38 * dt);
      } else {
        player.nitro = Math.min(100, player.nitro + 8 * dt);
      }
      nitroBar.style.width = `${player.nitro}%`;

      // Điểm số tăng theo quãng đường
      score += (currentSpeed * dt) * 0.06;
      const displayScore = Math.floor(score);
      scoreEl.textContent = displayScore;
      speedEl.textContent = `${Math.round(currentSpeed * 0.36)} KM/H`;

      // Đánh lái trái/phải
      let steer = 0;
      if (keys.left) steer -= 1;
      if (keys.right) steer += 1;

      player.x += steer * player.maxSteer * dt;
      player.x = Math.max(ROAD_LEFT + 4, Math.min(ROAD_RIGHT - player.w - 4, player.x));
      player.tilt += (steer * 0.14 - player.tilt) * Math.min(1, dt * 14);

      // Cuộn mặt đường
      roadOffset = (roadOffset + currentSpeed * dt) % 60;

      // Khói pô / lửa Nitro
      if (Math.random() < (isBoosting ? 0.95 : 0.35)) {
        addExhaustParticle(player.x + 12, player.y + player.h, isBoosting);
        addExhaustParticle(player.x + player.w - 12, player.y + player.h, isBoosting);
      }

      // Sinh xe địch
      spawnEnemyTimer += dt;
      const enemyInterval = Math.max(0.42, 1.15 - score * 0.00025);
      if (spawnEnemyTimer >= enemyInterval) {
        spawnEnemyTimer = 0;
        spawnEnemy();
      }

      // Sinh xu & bình Nitro
      spawnPickupTimer += dt;
      if (spawnPickupTimer >= 1.35) {
        spawnPickupTimer = 0;
        spawnPickup();
      }

      // Cập nhật xe địch
      for (let i = enemies.length - 1; i >= 0; i--) {
        const e = enemies[i];
        e.y += (currentSpeed - e.speedOffset) * dt;

        if (checkCollision(player, e, 6)) {
          gameOver();
          return;
        }

        if (e.y > H + 100) {
          enemies.splice(i, 1);
        }
      }

      // Cập nhật vật phẩm
      for (let i = pickups.length - 1; i >= 0; i--) {
        const p = pickups[i];
        p.y += currentSpeed * 0.85 * dt;
        p.pulse += dt * 6;

        if (checkCollision(player, p, 2)) {
          if (p.type === "COIN") {
            score += 50;
            playTone(880, "sine", 0.12, 0.09);
          } else {
            score += 100;
            player.nitro = Math.min(100, player.nitro + 45);
            playTone(620, "square", 0.18, 0.08);
          }
          pickups.splice(i, 1);
          continue;
        }

        if (p.y > H + 50) {
          pickups.splice(i, 1);
        }
      }
    }

    function gameOver() {
      gameState = "GAMEOVER";
      shakeTime = 0.4;
      addExplosion(player.x + player.w / 2, player.y + player.h / 2);
      playTone(130, "sawtooth", 0.45, 0.14);

      const finalScore = Math.floor(score);
      if (finalScore > highScore) {
        highScore = finalScore;
        localStorage.setItem("turbo_neon_highscore", highScore);
        highScoreEl.textContent = highScore;
      }

      setTimeout(() => {
        overlayTitle.textContent = "VA CHẠM RỒI!";
        overlayDesc.innerHTML = `Điểm của bạn: <b style="color:#38bdf8;font-size:20px;">${finalScore}</b><br>Kỷ lục cao nhất: <b style="color:#facc15;">${highScore}</b>`;
        startBtn.textContent = "CHƠI LẠI NGAY";
        overlay.classList.remove("hidden");
      }, 450);
    }

    function drawCar(x, y, w, h, bodyColor, roofColor, isPlayer = false, tilt = 0) {
      ctx.save();
      ctx.translate(x + w / 2, y + h / 2);
      ctx.rotate(tilt);
      ctx.translate(-w / 2, -h / 2);

      // Bóng xe
      ctx.fillStyle = "rgba(0, 0, 0, 0.35)";
      ctx.beginPath();
      ctx.roundRect(3, 6, w, h, 10);
      ctx.fill();

      // 4 bánh xe
      ctx.fillStyle = "#090d16";
      ctx.fillRect(-3, 12, 5, 16);
      ctx.fillRect(w - 2, 12, 5, 16);
      ctx.fillRect(-3, h - 28, 5, 16);
      ctx.fillRect(w - 2, h - 28, 5, 16);

      // Thân xe
      ctx.fillStyle = bodyColor;
      ctx.beginPath();
      ctx.roundRect(0, 0, w, h, 10);
      ctx.fill();

      // Kính chắn gió & mui xe
      ctx.fillStyle = "#0f172a";
      ctx.beginPath();
      ctx.roundRect(6, 16, w - 12, h - 34, 6);
      ctx.fill();

      ctx.fillStyle = roofColor;
      ctx.beginPath();
      ctx.roundRect(8, 26, w - 16, h - 48, 4);
      ctx.fill();

      // Đèn pha / đèn hậu
      if (isPlayer) {
        // Đèn pha chiếu sáng phía trước
        ctx.fillStyle = "#fef08a";
        ctx.fillRect(6, 2, 8, 4);
        ctx.fillRect(w - 14, 2, 8, 4);

        const grad = ctx.createLinearGradient(0, 0, 0, -75);
        grad.addColorStop(0, "rgba(254, 240, 138, 0.32)");
        grad.addColorStop(1, "rgba(254, 240, 138, 0)");
        ctx.fillStyle = grad;
        ctx.beginPath();
        ctx.moveTo(4, 2);
        ctx.lineTo(-14, -75);
        ctx.lineTo(w + 14, -75);
        ctx.lineTo(w - 4, 2);
        ctx.closePath();
        ctx.fill();
      } else {
        // Đèn hậu đỏ cho xe phía trước
        ctx.fillStyle = "#fca5a5";
        ctx.fillRect(6, h - 4, 8, 3);
        ctx.fillRect(w - 14, h - 4, 8, 3);
      }

      ctx.restore();
    }

    function render() {
      ctx.save();
      if (shakeTime > 0) {
        const mag = shakeTime * 18;
        ctx.translate((Math.random() - 0.5) * mag, (Math.random() - 0.5) * mag);
      }

      // Nền cỏ/lề đường hai bên
      ctx.fillStyle = "#0b1324";
      ctx.fillRect(0, 0, W, H);

      // Mặt đường nhựa
      ctx.fillStyle = "#1e293b";
      ctx.fillRect(ROAD_LEFT, 0, ROAD_WIDTH, H);

      // Vạch kẻ biên đường trắng - đỏ kiểu đường đua
      for (let y = -60 + roadOffset; y < H; y += 60) {
        ctx.fillStyle = "#ef4444";
        ctx.fillRect(ROAD_LEFT - 8, y, 8, 30);
        ctx.fillRect(ROAD_RIGHT, y, 8, 30);

        ctx.fillStyle = "#f8fafc";
        ctx.fillRect(ROAD_LEFT - 8, y + 30, 8, 30);
        ctx.fillRect(ROAD_RIGHT, y + 30, 8, 30);
      }

      // Vạch phân làn đường đứt khúc
      ctx.strokeStyle = "rgba(226, 232, 240, 0.32)";
      ctx.lineWidth = 3;
      ctx.setLineDash([28, 32]);
      ctx.lineDashOffset = -roadOffset;
      for (let i = 1; i < LANES; i++) {
        const lx = ROAD_LEFT + i * LANE_WIDTH;
        ctx.beginPath();
        ctx.moveTo(lx, 0);
        ctx.lineTo(lx, H);
        ctx.stroke();
      }
      ctx.setLineDash([]);

      // Vẽ vật phẩm
      for (const p of pickups) {
        const scale = 1 + Math.sin(p.pulse) * 0.1;
        ctx.save();
        ctx.translate(p.x + p.w / 2, p.y + p.h / 2);
        ctx.scale(scale, scale);

        if (p.type === "COIN") {
          ctx.fillStyle = "#facc15";
          ctx.beginPath();
          ctx.arc(0, 0, p.w / 2, 0, Math.PI * 2);
          ctx.fill();
          ctx.strokeStyle = "#fef08a";
          ctx.lineWidth = 2;
          ctx.stroke();

          ctx.fillStyle = "#854d0e";
          ctx.font = "bold 13px sans-serif";
          ctx.textAlign = "center";
          ctx.textBaseline = "middle";
          ctx.fillText("$", 0, 1);
        } else {
          ctx.fillStyle = "#38bdf8";
          ctx.beginPath();
          ctx.roundRect(-p.w / 2, -p.h / 2, p.w, p.h, 6);
          ctx.fill();
          ctx.fillStyle = "#0f172a";
          ctx.font = "bold 14px sans-serif";
          ctx.textAlign = "center";
          ctx.textBaseline = "middle";
          ctx.fillText("⚡", 0, 1);
        }
        ctx.restore();
      }

      // Vẽ hạt khói/lửa
      for (const pt of particles) {
        ctx.globalAlpha = Math.max(0, pt.life / pt.maxLife);
        ctx.fillStyle = pt.color;
        ctx.beginPath();
        ctx.arc(pt.x, pt.y, pt.size, 0, Math.PI * 2);
        ctx.fill();
      }
      ctx.globalAlpha = 1;

      // Vẽ xe địch
      for (const e of enemies) {
        drawCar(e.x, e.y, e.w, e.h, e.color.body, e.color.roof, false, 0);
      }

      // Vẽ xe người chơi
      if (gameState !== "GAMEOVER") {
        drawCar(player.x, player.y, player.w, player.h, "#38bdf8", "#0369a1", true, player.tilt);
      }

      ctx.restore();
    }

    let lastTime = performance.now();
    function gameLoop(now) {
      const dt = Math.min(0.05, (now - lastTime) / 1000);
      lastTime = now;

      update(dt);
      render();
      requestAnimationFrame(gameLoop);
    }

    requestAnimationFrame(gameLoop);
  </script>
